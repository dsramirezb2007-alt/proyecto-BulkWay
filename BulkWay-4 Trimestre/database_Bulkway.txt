CREATE DATABASE IF NOT EXISTS bulkway_db;
USE bulkway_db;
##crear tablas
CREATE TABLE Roles (
    id_Rol INT AUTO_INCREMENT PRIMARY KEY,
    Nombre_Rol VARCHAR(40) NOT NULL
);

CREATE TABLE TipoDocumentos (
    idTipo_Documento INT AUTO_INCREMENT PRIMARY KEY,
    Nombre_Documento VARCHAR(45) NOT NULL
);

CREATE TABLE Usuarios (
    id_Usuario INT AUTO_INCREMENT PRIMARY KEY,
    id_Rol INT NOT NULL,
    id_Tipo_Documento INT NOT NULL,
    Correo_Usuario VARCHAR(200) NOT NULL UNIQUE,
    Contraseña_Usuario VARCHAR(200) NOT NULL,
    Estado_Usuario ENUM('Activo', 'Inactivo', 'Suspendido', 'Eliminado', 'Cancelado', 'Pendiente') NOT NULL DEFAULT 'Activo',
    nombre VARCHAR(100) NOT NULL,
    telefono VARCHAR(45) NOT NULL,
    FOREIGN KEY (id_Rol) REFERENCES Roles(id_Rol),
    FOREIGN KEY (id_Tipo_Documento) REFERENCES TipoDocumentos(idTipo_Documento) -- Corregido aquí
);

CREATE TABLE Rutas (
    idRuta INT AUTO_INCREMENT PRIMARY KEY,
    nombre_ruta VARCHAR(100) NOT NULL
);

CREATE TABLE Vehiculos (
    id_Vehiculo INT AUTO_INCREMENT PRIMARY KEY,
    modelo VARCHAR(60) NOT NULL,
    placa VARCHAR(20) NOT NULL UNIQUE,
    color VARCHAR(25) NOT NULL,
    tipo_vehiculo ENUM('Camión', 'Furgón', 'Camioneta', 'Motocarro') NOT NULL
);

CREATE TABLE CategoriaProductos (
    id_CategoriaProducto INT AUTO_INCREMENT PRIMARY KEY,
    Nombre_Categoria VARCHAR(100) NOT NULL,
    Descripcion VARCHAR(200) NOT NULL
);

CREATE TABLE Productos (
    id_Producto INT AUTO_INCREMENT PRIMARY KEY,
    Nombre_Producto VARCHAR(100) NOT NULL,
    Descripcion VARCHAR(200) NOT NULL,
    Precio_Producto DECIMAL(10,2) NOT NULL,
    Stock_Producto INT NOT NULL,
    id_CategoriaProducto INT NOT NULL,
    FOREIGN KEY (id_CategoriaProducto) REFERENCES CategoriaProductos(id_CategoriaProducto)
);

CREATE TABLE pedidos (
    id_Pedido INT AUTO_INCREMENT PRIMARY KEY,
    Fecha_Pedido DATETIME NOT NULL,
    Estado_Pedido ENUM('Pendiente', 'Asignado', 'En transito', 'Completado') NOT NULL DEFAULT 'Pendiente',
    descuento DECIMAL(12,2) DEFAULT 0.00,
    costo_productos DECIMAL(12,2) NOT NULL,
    fecha_entrega DATE NOT NULL,
    fecha_creacion DATETIME NOT NULL,
    id_Cliente INT NOT NULL,
    FOREIGN KEY (id_Cliente) REFERENCES Usuarios(id_Usuario)
);

CREATE TABLE DetallePedidos (
    id_Detalle_Pedido INT AUTO_INCREMENT PRIMARY KEY,
    Cantidad_Pedido INT NOT NULL,
    id_Pedido INT NOT NULL,
    id_Producto INT NOT NULL,
    FOREIGN KEY (id_Pedido) REFERENCES pedidos(id_Pedido),
    FOREIGN KEY (id_Producto) REFERENCES Productos(id_Producto)
);

CREATE TABLE ReciboCajas (
    id_Recibo_Caja INT AUTO_INCREMENT PRIMARY KEY,
    Fecha_Factura DATETIME NOT NULL,
    Estado ENUM('Pago', 'Pendiente', 'Cancelado') NOT NULL DEFAULT 'Pendiente',
    impuesto DECIMAL(5,2) NOT NULL,
    legalizacion VARCHAR(100) NOT NULL,
    fecha_vencimiento DATETIME NOT NULL,
    total_pagar DECIMAL(10,2) NOT NULL,
    id_Pedido INT NOT NULL,
    FOREIGN KEY (id_Pedido) REFERENCES pedidos(id_Pedido)
);

CREATE TABLE Envios (
    id_Envio INT AUTO_INCREMENT PRIMARY KEY,
    id_Pedido INT NOT NULL,
    Empleado INT NOT NULL, 
    Fecha_Salida DATETIME NOT NULL,
    Fecha_Entrega DATETIME NOT NULL,
    Estado_Envio ENUM('En Bodega', 'En Ruta', 'Entregado', 'Cancelado') NOT NULL DEFAULT 'En Bodega',
    Cod_SeguimientoEnvio VARCHAR(100) NOT NULL UNIQUE,
    id_Usuario INT NOT NULL, 
    id_Factura INT NOT NULL,
    id_Ruta INT NOT NULL,
    id_Vehiculo INT NOT NULL,
    FOREIGN KEY (id_Pedido) REFERENCES pedidos(id_Pedido),
    FOREIGN KEY (Empleado) REFERENCES Usuarios(id_Usuario),
    FOREIGN KEY (id_Usuario) REFERENCES Usuarios(id_Usuario),
    FOREIGN KEY (id_Factura) REFERENCES ReciboCajas(id_Recibo_Caja),
    FOREIGN KEY (id_Ruta) REFERENCES Rutas(idRuta),
    FOREIGN KEY (id_Vehiculo) REFERENCES Vehiculos(id_Vehiculo)
);

CREATE TABLE MovimientoInventarios (
    id_MovimientoInventario INT AUTO_INCREMENT PRIMARY KEY,
    Tipo_Movimiento ENUM('Entrada', 'Salida', 'Ajuste interno') NOT NULL,
    FechaHora_Movimiento DATETIME NOT NULL,
    Descripcion VARCHAR(200) NOT NULL
);

CREATE TABLE DetalleMovimientos (
    idDetalle_Movimiento INT AUTO_INCREMENT PRIMARY KEY,
    Tipo_Movimiento ENUM('Entrada', 'Salida', 'Ajuste interno') NOT NULL,
    cantidad INT NOT NULL,
    precio_unitario DECIMAL(12,2) NOT NULL,
    porcentaje_impuesto DECIMAL(5,2) NOT NULL,
    estado_producto ENUM('Disponible', 'Dañado', 'Reservado') NOT NULL,
    numero_lote VARCHAR(100) NOT NULL,
    id_MovimientoInventario INT NOT NULL,
    id_Producto INT NOT NULL,
    FOREIGN KEY (id_MovimientoInventario) REFERENCES MovimientoInventarios(id_MovimientoInventario),
    FOREIGN KEY (id_Producto) REFERENCES Productos(id_Producto)
);
